import CoreGraphics
import CoreText
import Foundation
import ImageIO
import UniformTypeIdentifiers

// Her satır sütunlara bölünür: (x konumu çarpanı, metin)
let header = "HARCAMALARINIZ - Bonus kazanın!"
let rows: [[String]] = [
    ["16 Ekim 2026", "EXAMPLE SIGORTA", "1.878,96x9=16.910,00 1.Taksit", "8,46", "1.878,96"],
    ["18 Ekim 2026", "EXAMPLE SHOP", "254,31x3=762,89 1.Taksit", "0,15", "254,31"],
    ["30 Eylül 2026", "EXAMPLE MARKET", "1.194,50x2=2.389,00 2.Taksit", "", "1.194,50"],
]
let summary = [
    ["Dönem Borcunuz", "46.354,97 TL"],
    ["Asgari Ödeme Tutarı", "9.270,99 TL"],
    ["Son Ödeme Tarihi", "02 Kasım 2026"],
]
let out = URL(fileURLWithPath: CommandLine.arguments[1])

func text(_ ctx: CGContext, _ string: String, x: CGFloat, y: CGFloat, size: CGFloat) {
    let attr = NSAttributedString(string: string, attributes: [
        kCTFontAttributeName as NSAttributedString.Key: CTFontCreateWithName("Helvetica" as CFString, size, nil),
        kCTForegroundColorFromContextAttributeName as NSAttributedString.Key: true,
    ])
    ctx.textPosition = CGPoint(x: x, y: y)
    CTLineDraw(CTLineCreateWithAttributedString(attr), ctx)
}

func draw(_ ctx: CGContext, width: CGFloat, height: CGFloat, size: CGFloat) {
    let columns: [CGFloat] = [0.03, 0.22, 0.42, 0.78, 0.88].map { $0 * width }
    let step = size * 2.4
    var y = height - size * 2
    // Başlık satırların hemen üstünde; çıkarma bazen onu ilk satırla birleştirir.
    text(ctx, header, x: columns[0], y: y, size: size)
    y -= step
    for row in rows {
        for (index, cell) in row.enumerated() where !cell.isEmpty {
            text(ctx, cell, x: columns[index], y: y, size: size)
        }
        y -= step
    }
    y -= step
    for row in summary {
        text(ctx, row[0], x: columns[0], y: y, size: size)
        text(ctx, row[1], x: columns[2], y: y, size: size)
        y -= step
    }
}

var box = CGRect(x: 0, y: 0, width: 800, height: 260)
let pdf = CGContext(out.appendingPathComponent("statement-columns.pdf") as CFURL, mediaBox: &box, nil)!
pdf.beginPDFPage(nil)
pdf.setFillColor(CGColor(gray: 0, alpha: 1))
draw(pdf, width: 800, height: 260, size: 11)
pdf.endPDFPage()
pdf.closePDF()

let w = 2400, h = 1000
let ctx = CGContext(data: nil, width: w, height: h, bitsPerComponent: 8, bytesPerRow: 0,
                    space: CGColorSpaceCreateDeviceGray(), bitmapInfo: CGImageAlphaInfo.none.rawValue)!
ctx.setFillColor(CGColor(gray: 1, alpha: 1)); ctx.fill(CGRect(x: 0, y: 0, width: w, height: h))
ctx.setFillColor(CGColor(gray: 0, alpha: 1))
draw(ctx, width: CGFloat(w), height: CGFloat(h), size: 44)
let dest = CGImageDestinationCreateWithURL(out.appendingPathComponent("statement-columns.png") as CFURL, UTType.png.identifier as CFString, 1, nil)!
CGImageDestinationAddImage(dest, ctx.makeImage()!, nil)
CGImageDestinationFinalize(dest)
